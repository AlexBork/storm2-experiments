%% #1: csv file, relative to \resultsdir  #2: what to show if it exists
\newcommand{\ifresults}[2]{%
  \IfFileExists{\resultsdir/#1}{#2}{\emph{No results in \texttt{\resultsdir/#1}.}\par}}

%% #1: csv file, relative to \resultsdir  #2: column
%% #3: what to show if the file has the column  #4: otherwise
\newread\headerfile
\makeatletter
\newcommand{\ifcolumn}[4]{%
  \begingroup
    \endlinechar=-1
    \openin\headerfile=\resultsdir/#1
    \read\headerfile to \header
    \closein\headerfile
    \edef\header{,\header,}%
    \in@{,#2,}{\header}%
  \expandafter\endgroup\ifin@
    \expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi{#3}{#4}}
\makeatother
